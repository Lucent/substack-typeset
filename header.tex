% === Page layout (memoir native, replaces geometry package) ===
% Stock = physical paper; trimmed = final page after cutting
\setstocksize{9.25in}{6.25in}
\settrimmedsize{9in}{6in}{*}
\settrims{0.125in}{0.125in}
% Margins: inner (spine) + binding offset, outer (edge)
\setbinding{0.25in}
\setlrmarginsandblock{0.75in}{0.5in}{*}
\setulmarginsandblock{0.625in}{0.75in}{*}
\checkandfixthelayout
\raggedbottom

% === Line spacing (memoir native, replaces setspace package) ===
\setSingleSpace{1.15}
\SingleSpacing

% === Half-title page (injected before the full title page via \pretitle) ===
\pretitle{%
  \thispagestyle{empty}%
  \vspace*{\fill}%
  {\centering\LARGE\itshape Surviving Scale\par}%
  \vspace*{\fill}%
  \cleardoublepage
  \thispagestyle{empty}%
  \vspace*{\fill}%
  \begin{center}\HUGE\bfseries}
\posttitle{\par\end{center}}
\preauthor{\begin{center}\Large\vspace{0.5em}}
\postauthor{\par\end{center}\vspace*{\fill}}
\predate{\begin{center}\large}
\postdate{\par\end{center}\vspace{1cm}}

% === Chapter style ===
% 'article' — no "Chapter N" prefix, just the title. Try others from
% the MemoirChapStyles catalog: komalike, dowding, tandh, madsen, etc.
\chapterstyle{article}
% Fix precis shift — memoir defaults to -2\baselineskip unless 'article' class
% option is set, but we're only using the article *chapter style*, not the option.
\prechapterprecisshift=0pt

% === Suppress section numbering and section marks in running headers ===
\setsecnumdepth{chapter}
\renewcommand{\sectionmark}[1]{}

% === Part style (memoir native, replaces titlesec) ===
% Suppresses "Part N" prefix, allows content (epigraphs) after title
\renewcommand{\printpartname}{}
\renewcommand{\printpartnum}{}
\renewcommand{\thepart}{}
\renewcommand{\parttitlefont}{\centering\bfseries\Huge}
\renewcommand{\beforepartskip}{\vspace*{3cm}}
\renewcommand{\afterpartskip}{\vspace{1cm}}

% === Epigraphs (memoir built-in, no package needed) ===
%\setlength{\epigraphwidth}{0.8\textwidth}

% === TOC styling (memoir built-in) ===
% Prevent TOC from listing itself (memoir adds it by default; Pandoc calls
% the unstarred \tableofcontents which we can't control, so redirect it)
\let\oldtableofcontents\tableofcontents
\renewcommand{\tableofcontents}{\oldtableofcontents*}
% Parts: bold, spaced, no page numbers (they're section dividers, not destinations)
\renewcommand{\cftpartfont}{\bfseries\large}
\renewcommand{\cftpartleader}{}
\cftpagenumbersoff{part}
\setlength{\cftbeforepartskip}{1em}
% Chapters: normal weight, dot leaders, tight spacing
% Discourage page breaks between chapters so TOC breaks at part boundaries
\renewcommand{\cftchapterbreak}{\nobreak}
\renewcommand{\cftchapterfont}{\normalfont}
\renewcommand{\cftchapterpagefont}{\normalfont}
\renewcommand{\cftchapterleader}{\cftdotfill{\cftdotsep}}
\setlength{\cftbeforechapterskip}{0.3em}

% === Per-chapter footnote reset (plain numbers, no chapter prefix) ===
\counterwithin{footnote}{chapter}
\renewcommand{\thefootnote}{\arabic{footnote}}

% === Scene breaks: ornamental asterisks instead of horizontal rules ===
% Pandoc renders --- as \rule; memoir's \pfbreak gives centered * * *
\renewcommand{\pfbreakskip}{1.5em}

% === Ragged-right footnotes (memoir native + ragged2e) ===
\usepackage{ragged2e}
\renewcommand{\foottextfont}{\footnotesize\RaggedRight}

% === Drop caps ===
\usepackage{lettrine}

% === Fallback font for blackletter W (Wikipedia marker) ===
\usepackage{newunicodechar}
\newfontfamily\mathfallback{Latin Modern Math}
\newunicodechar{𝔚}{{\mathfallback 𝔚}}
